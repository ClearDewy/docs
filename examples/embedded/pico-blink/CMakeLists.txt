cmake_minimum_required(VERSION 3.13)
if(NOT DEFINED PICO_SDK_PATH)
  message(FATAL_ERROR "Pass -DPICO_SDK_PATH=/absolute/path/to/pico-sdk (2.1.1)")
endif()
include(${PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(learning_blink C CXX ASM)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
pico_sdk_init()
if(NOT PICO_BOARD STREQUAL "pico")
  message(FATAL_ERROR "This lab targets Raspberry Pi Pico RP2040, PICO_BOARD=pico (not W or Pico 2)")
endif()
add_executable(learning_blink main.c)
target_link_libraries(learning_blink pico_stdlib)
pico_enable_stdio_usb(learning_blink 0)
pico_enable_stdio_uart(learning_blink 0)
pico_add_extra_outputs(learning_blink)
